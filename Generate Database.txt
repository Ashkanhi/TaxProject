CREATE PROCEDURE [dbo].[TaxGovermentGetInvoiceDetailForSend]
(
    @InvoiceID NVARCHAR(50) = '',
    @IsReturn BIT = 0
)
AS
BEGIN

    DECLARE @Fee DECIMAL(18, 0),
            @DiscountP INT,
            @PriceAfterDiscount DECIMAL(18, 0),
            @Qty DECIMAL(18, 3);



    SELECT FTR.fldFactorID,
           FTR.fldID,
           FTR.fldRowNo,
           DiscountPercent = 0,
           Discount = FTR.fldDiscountPrice,
           Qty = (FTR.fldQty),
           Fee = FTR.fldPrice / FTR.fldQty,
           FullPrice = FTR.fldPrice,
           TaxPrice = SUM(FTRT.fldPrice) OVER (PARTITION BY FTR.fldID)
    INTO #tmpCalcItems
    FROM Sales.tbl_FactorRow FTR
        INNER JOIN
        (
            SELECT fldFactorRowID,
                   SUM(fldPrice) fldPrice
            FROM Sales.tbl_FactorRowTaxes
            GROUP BY fldFactorRowID
        ) FTRT
            ON FTRT.fldFactorRowID = FTR.fldID
    WHERE FTR.fldFactorID = @InvoiceID;




    SELECT Sstid = Item.fldComponentDescription,                                       --ifmv.Barcode,
           Sstt = Item.fldDescription,
           Mu = ISNULL(Unit.fldComponentDescription, N'1027'),
           Am = CI.Qty,                                                                -- تعداد
           Fee = CONVERT(DECIMAL(18, 0), CI.Fee),                                      --- Fee,
           Prdis = CONVERT(DECIMAL(18, 0), CI.FullPrice),                              --مبلغ قبل تخفیف
           Dis = CONVERT(DECIMAL(18, 0), CI.Discount),                                 --مبلغ تخفیف
           Adis = CONVERT(DECIMAL(18, 0), CI.FullPrice - CI.Discount),          --مبلغ بعد از تخفیف
           Vra = CONVERT(   DECIMAL(18, 2),
                            (CASE
                                 WHEN CI.TaxPrice > 0 THEN
                                     9 --0.09
                                 ELSE
                                     0
                             END
                            )
                        ),                                                             --نرخ مالیات بر ارزش افزوده
           Vam = CONVERT(DECIMAL(18, 0), CI.TaxPrice),                                 --مبلغ مالیات بر ارزش افزوده                                                --FTR.TaxAmount + FTR.TollAmount,
           Odt = '',
           Odr = CONVERT(DECIMAL, 0),
           Odam = CONVERT(DECIMAL, 0),
           Olt = '',
           Olr = CONVERT(DECIMAL, 0),
           Olam = CONVERT(DECIMAL, 0),
           Consfee = CONVERT(DECIMAL, 0),
           Spro = CONVERT(DECIMAL, 0),
           Bros = CONVERT(DECIMAL, 0),
           Tcpbs = CONVERT(DECIMAL, 0),
           Cop = CONVERT(DECIMAL, 0),
                                                                                       --سهم نقدی از پرداخت
           Vop = CONVERT(DECIMAL(18, 0), 0),                                           --سهم ارزش افزوده از پرداخت                                                --FTR.TaxAmount + FTR.TollAmount,
           Bsrn = '0',
           Tsstam = CONVERT(DECIMAL(18, 0), CI.FullPrice - CI.Discount + CI.TaxPrice), --مبلغ کل کالا و خدمات
           Nw = CONVERT(DECIMAL, 0),
           Ssrv = CONVERT(DECIMAL, 0),
           Sscv = CONVERT(DECIMAL, 0),
           FT.fldID,
           GETDATE() [Date]
    INTO #tmp
    FROM Sales.tbl_FactorTitle AS FT
        INNER JOIN Sales.tbl_FactorRow AS FTR
            ON FT.fldID = FTR.fldFactorID
        INNER JOIN #tmpCalcItems CI
            ON FTR.fldID = CI.fldID
        INNER JOIN Common.tbl_CodingComponents P
            ON FT.fldCustomerID = P.fldformGeneratorID
        INNER JOIN Common.tbl_Persons Pers
            ON P.fldformGeneratorID = Pers.fldID
        INNER JOIN Common.tbl_CodingComponents Item
            ON FTR.fldItemID = Item.fldformGeneratorID
        LEFT JOIN Common.tbl_CodingComponentItem CItm
            ON FTR.fldItemID = CItm.fldformGeneratorID
        LEFT JOIN Common.tbl_CodingComponents Unit
            ON CItm.fldUnit = Unit.fldformGeneratorID
    WHERE FT.fldID = @InvoiceID;


    INSERT INTO TaxGovermentInoiceDetailLog
    SELECT *
    FROM #tmp;
    SELECT *
    FROM #tmp;
    DROP TABLE #tmp;

END;
GO

        SELECT @TaxID = TaxId
        FROM dbo.TaxGovermentInvoiceFinal
        WHERE InvoiceId = @InvoiceId;

    IF (@InvoiceType <> 3)
       AND LEN(ISNULL(@InvoiceId, '')) = 0

        SELECT TOP 1
               @InvoiceId = FT.fldID,
               @CustomerId = FT.fldCustomerID,
               @IsReturn = (CASE
                                WHEN FT.fldDocumentType = 2 THEN
                                    0
                                ELSE
                                    1
                            END

            INNER JOIN Sales.tbl_FactorRow AS FTR

                ON FT.fldCustomerID = Per.fldformGeneratorID

                ON CI.fldformGeneratorID = FTR.fldItemID
            LEFT JOIN dbo.TaxGovermentInvoiceFinal TGov
                ON TGov.InvoiceId = FT.fldID

              AND FT.fldState = 1

              (
                  TGov.InvoiceId IS NULL
                  OR TGov.Done = 0

           Fee = FTR.fldPrice / FTR.fldQty,

CREATE PROCEDURE [dbo].[TaxGovermentInvoiceFinalInsertUpdate]
    @InvoiceId NVARCHAR(50),
    @LastStatus NVARCHAR(100),
    @Done BIT,
    @TaxId NVARCHAR(100),
    @LastRefrenceId NVARCHAR(50),
	@Cancel bit = 0
AS
BEGIN

    SET NOCOUNT ON;
	IF @Cancel = 1
	Begin
		Delete From [dbo].[TaxGovermentInvoiceFinal] WHere InvoiceId = @InvoiceId + '#R'
		INSERT INTO [dbo].[TaxGovermentInvoiceFinal]
            (
                InvoiceId,
                LastStatus,
                Done,
                TaxId,
                LastRefrenceId,
                LastResponseCheckTime
            )
            SELECT @InvoiceId + '#R',
                   'Cancel',
                   @Done,
                   @TaxId,
                   @LastRefrenceId,
                   GETDATE();
			UPDATE [dbo].[TaxGovermentInvoiceFinal]
					SET LastStatus = 'Cancel',
						Done = @Done,
						LastRefrenceId = @LastRefrenceId,
						LastResponseCheckTime = GETDATE()
					WHERE InvoiceId = @InvoiceId;
	End
	Else
	Begin
		IF @InvoiceId IS NULL
       OR @InvoiceId = ''
       OR LEN(@InvoiceId) = 0
    BEGIN
		PRINT'InvoiceId IS NULL'
        UPDATE [dbo].[TaxGovermentInvoiceFinal]
        SET LastStatus = @LastStatus,
            Done = @Done,
            LastRefrenceId = @LastRefrenceId,
			LastResponseCheckTime = GETDATE()
        WHERE LastRefrenceId = @LastRefrenceId;
    END;
    ELSE
    BEGIN
        IF NOT EXISTS
        (
            SELECT InvoiceId
            FROM [dbo].[TaxGovermentInvoiceFinal]
            WHERE InvoiceId = @InvoiceId
        )
        BEGIN
            PRINT 'Insert';
            INSERT INTO [dbo].[TaxGovermentInvoiceFinal]
            (
                InvoiceId,
                LastStatus,
                Done,
                TaxId,
                LastRefrenceId,
                LastResponseCheckTime
            )
            SELECT @InvoiceId,
                   @LastStatus,
                   @Done,
                   @TaxId,
                   @LastRefrenceId,
                   GETDATE();
        END;
        ELSE
            PRINT 'Update';
        UPDATE [dbo].[TaxGovermentInvoiceFinal]
        SET LastStatus = @LastStatus,
            Done = @Done,
            LastRefrenceId = @LastRefrenceId,
            TaxId = @TaxId,
            LastResponseCheckTime = GETDATE()
        WHERE InvoiceId = @InvoiceId;
    END;
	ENd
END;

GO

CREATE PROCEDURE [dbo].[TaxGovermentLastRefrenceId]
AS
BEGIN
    DECLARE @time INT;
    SELECT @time = ISNULL(ResponseCheckerTimeInSec, 300)
    FROM dbo.TaxGovermentSettingInfo;

    SELECT TOP 1
           LastRefrenceId
    FROM dbo.TaxGovermentInvoiceFinal
    WHERE LOWER(LastStatus) IN ( 'pending', 'sent' )
          AND
          (
              DATEADD(SECOND, @time, LastResponseCheckTime) < GETDATE()
              OR LastResponseCheckTime IS NULL
          );
END;

CREATE PROCEDURE [dbo].[TaxGovermentSaveInvoiceLogInsert]
@ReferenceNumber nvarchar(100),@Status nvarchar(50),@UId nvarchar(100),@Data nvarchar(Max),@PacketType nvarchar(50)
AS
BEGIN
	SET NOCOUNT ON;

	Declare @TaxGovermentInvoiceInfoId bigint;
	Select @TaxGovermentInvoiceInfoId = TaxGovermentInvoiceInfoId From [dbo].[TaxGovermentInvoiceInfo] Where ReferenceNumber = @ReferenceNumber;

	Insert Into [dbo].[TaxGovermentSendLog](TaxGovermentInvoiceInfoId,RefrenceId,[Status],[UId],[Date],Json,PacketType)
		Select @TaxGovermentInvoiceInfoId,@ReferenceNumber,@Status,@UId,Getdate(),@Data,@PacketType
END
